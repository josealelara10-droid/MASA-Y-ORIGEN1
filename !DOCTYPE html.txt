    <script>
        let cart = [];

        function addToCart(name, price) {
            cart.push({ name, price });
            updateCart();
            toggleCart();
        }

        function updateCart() {
            const countEl = document.getElementById('cart-count');
            const itemsEl = document.getElementById('cart-items');
            const totalEl = document.getElementById('cart-total');
            
            countEl.innerText = cart.length;

            if (cart.length === 0) {
                itemsEl.innerHTML = '<p class="text-xs text-stone-500 text-center py-8">Tu carrito está vacío actualmente.</p>';
                totalEl.innerText = '$0';
                return;
            }

            let total = 0;
            itemsEl.innerHTML = cart.map((item, index) => {
                total += item.price;
                return `
                    <div class="flex justify-between items-center bg-warm-50 p-3 rounded-xl text-xs">
                        <div>
                            <p class="font-bold text-amber-950">${item.name}</p>
                            <p class="text-amber-700 font-semibold">$${item.price.toLocaleString()}</p>
                        </div>
                        <button onclick="removeFromCart(${index})" class="text-red-500 hover:text-red-700 font-bold">Quitar</button>
                    </div>
                `;
            }).join('');

            totalEl.innerText = '$' + total.toLocaleString();
        }

        function removeFromCart(index) {
            cart.splice(index, 1);
            updateCart();
        }

        function toggleCart() {
            const modal = document.getElementById('cart-modal');
            modal.classList.toggle('hidden');
        }

        function checkout() {
            if(cart.length === 0) {
                alert("Tu carrito está vacío");
                return;
            }
            alert("¡Gracias por tu pedido! Serás redirigido a la pasarela de pago segura.");
            cart = [];
            updateCart();
            toggleCart();
        }

        function calculateSubscription() {
            const qty = parseInt(document.getElementById('sub-quantity').value);
            document.getElementById('sub-quantity-val').innerText = qty;
            const pastry = document.getElementById('sub-pastry').checked;
            
            let basePrice = qty * 15000;
            if(pastry) basePrice += 15000;

            document.getElementById('sub-total').innerHTML = `$${basePrice.toLocaleString()} <span class="text-xs text-amber-400/80 font-normal">/mes</span>`;
        }

        function confirmSubscription() {
            alert("¡Excelente elección! Te redirigiremos para completar los datos de entrega de tu suscripción.");
        }

        function filterProducts(category) {
            const cards = document.querySelectorAll('.product-card');
            cards.forEach(card => {
                if(category === 'todos' || card.classList.contains(category)) {
                    card.style.display = 'flex';
                } else {
                    card.style.display = 'none';
                }
            });
        }
    </script>
